<script>
document.addEventListener('DOMContentLoaded', function() {
    // Hide preloader
    setTimeout(() => {
        document.getElementById('preloader').classList.add('hidden');
    }, 500);

    const lockReasonModal = new bootstrap.Modal(document.getElementById('lockReasonModal'));
    const successToast = new bootstrap.Toast(document.getElementById('successToast'));
    let currentTopicId = null;
    let currentAction = null;

    // Course filter functionality
    const courseFilter = document.getElementById('courseFilter');
    courseFilter.addEventListener('change', function() {
        const selectedCourseId = this.value;
        const accordionItems = document.querySelectorAll('.accordion-item');
        
        accordionItems.forEach(item => {
            if (selectedCourseId === '' || item.dataset.courseId === selectedCourseId) {
                item.style.display = '';
                item.style.animation = 'fadeInScale 0.3s ease-out';
            } else {
                item.style.display = 'none';
            }
        });
        
        updateSelectedCount();
    });

    // Refresh button
    document.getElementById('refreshBtn').addEventListener('click', function() {
        this.classList.add('loading');
        this.innerHTML = '<div class="loading-spinner"></div><span>Refreshing...</span>';
        setTimeout(() => {
            location.reload();
        }, 500);
    });

    // Topic checkbox functionality
    document.querySelectorAll('.topic-checkbox').forEach(checkbox => {
        checkbox.addEventListener('change', function() {
            updateSelectedCount();
            updateBulkActionsBar();
        });
    });

    // Individual lock/unlock buttons
    document.querySelectorAll('.lock-topic').forEach(btn => {
        btn.addEventListener('click', function() {
            currentTopicId = this.dataset.topicId;
            currentAction = 'lock';
            lockReasonModal.show();
        });
    });

    document.querySelectorAll('.unlock-topic').forEach(btn => {
        btn.addEventListener('click', function() {
            const topicId = this.dataset.topicId;
            unlockTopic(topicId);
        });
    });

    // Bulk actions
    document.getElementById('bulkLockBtn').addEventListener('click', function() {
        const selectedTopics = getSelectedTopics();
        if (selectedTopics.length === 0) {
            showToast('Please select topics to lock.', 'warning');
            return;
        }
        currentAction = 'bulkLock';
        lockReasonModal.show();
    });

    document.getElementById('bulkUnlockBtn').addEventListener('click', function() {
        const selectedTopics = getSelectedTopics();
        if (selectedTopics.length === 0) {
            showToast('Please select topics to unlock.', 'warning');
            return;
        }
        bulkUnlockTopics(selectedTopics);
    });

    document.getElementById('clearSelection').addEventListener('click', function() {
        document.querySelectorAll('.topic-checkbox:checked').forEach(checkbox => {
            checkbox.checked = false;
        });
        updateSelectedCount();
        updateBulkActionsBar();
    });

    // Confirm lock button
    document.getElementById('confirmLock').addEventListener('click', function() {
        const reason = document.getElementById('lockReason').value;
        
        if (currentAction === 'lock' && currentTopicId) {
            lockTopic(currentTopicId, reason);
        } else if (currentAction === 'bulkLock') {
            const selectedTopics = getSelectedTopics();
            bulkLockTopics(selectedTopics, reason);
        }
        
        lockReasonModal.hide();
        document.getElementById('lockReason').value = '';
    });

    // Utility functions
    function updateSelectedCount() {
        const selectedCheckboxes = document.querySelectorAll('.topic-checkbox:checked');
        const selectedCount = selectedCheckboxes.length;
        document.getElementById('selectedCount').textContent = selectedCount;
        
        // Update bulk actions bar visibility
        const bulkActionsBar = document.getElementById('bulkActionsBar');
        if (selectedCount > 0) {
            bulkActionsBar.style.display = 'block';
            bulkActionsBar.classList.add('animate-fade-in-scale');
        } else {
            bulkActionsBar.style.display = 'none';
        }
    }

    function updateBulkActionsBar() {
        const selectedCount = getSelectedTopics().length;
        const bulkActionsBar = document.getElementById('bulkActionsBar');
        
        if (selectedCount > 0) {
            bulkActionsBar.style.display = 'block';
            bulkActionsBar.classList.add('animate-fade-in-scale');
        } else {

    function showToast(message, type = 'success') {
        const toastMessage = document.getElementById('toastMessage');
        const toastHeader = document.querySelector('#successToast .toast-header');
        
        toastMessage.textContent = message;
        
        // Update toast styling based on type
        if (type === 'error') {
            toastHeader.className = 'toast-header error';
        } else if (type === 'warning') {
            toastHeader.className = 'toast-header warning';
        } else {
            toastHeader.className = 'toast-header';
        }
        
        successToast.show();
    }

    function lockTopic(topicId, reason) {
        const button = document.querySelector(`[data-topic-id="${topicId}"]`);
        if (button) {
            button.disabled = true;
            button.innerHTML = '<div class="loading-spinner"></div><span>Locking...</span>';
        }

        fetch(`/topic-locking/${topicId}/lock`, {
            method: 'POST',
            headers: {
                'Content-Type': 'application/json',
                'X-CSRF-TOKEN': document.querySelector('meta[name="csrf-token"]').getAttribute('content')
            },
            body: JSON.stringify({ reason: reason })
        })
        .then(response => {
            if (!response.ok) {
                throw new Error(`HTTP error! status: ${response.status}`);
            }
            return response.json();
        })
        .then(data => {
            if (data.success) {
                showToast('Topic locked successfully!');
                setTimeout(() => location.reload(), 1500);
            } else {
                showToast('Error: ' + (data.message || 'Failed to lock topic'), 'error');
                resetButton(button, 'lock');
            }
        })
        .catch(error => {
            console.error('Error:', error);
            showToast('Network error occurred. Please try again.', 'error');
            resetButton(button, 'lock');
        });
    }

    function unlockTopic(topicId) {
        const button = document.querySelector(`[data-topic-id="${topicId}"]`);
        if (button) {
            button.disabled = true;
            button.innerHTML = '<div class="loading-spinner"></div><span>Unlocking...</span>';
        }

        fetch(`/topic-locking/${topicId}/unlock`, {
            method: 'POST',
            headers: {
                'Content-Type': 'application/json',
                'X-CSRF-TOKEN': document.querySelector('meta[name="csrf-token"]').getAttribute('content')
            }
        })
        .then(response => response.json())
        .then(data => {
            if (data.success) {
                showToast('Topic unlocked successfully!');
                setTimeout(() => location.reload(), 1000);
            } else {
                showToast('Error: ' + data.message, 'error');
                resetButton(button, 'unlock');
            }
        })
        .catch(error => {
            console.error('Error:', error);
            showToast('An error occurred while unlocking the topic.', 'error');
            resetButton(button, 'unlock');
        });
    }

    function resetButton(button, action) {
        if (!button) return;
        
        button.disabled = false;
        if (action === 'lock') {
            button.innerHTML = '<i class="fas fa-lock"></i><span>Lock</span>';
        } else if (action === 'unlock') {
            button.innerHTML = '<i class="fas fa-unlock"></i><span>Unlock</span>';
        }

        if (!button) return;
