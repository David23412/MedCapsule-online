## Part 2: Server Setup (Step-by-Step)

### Step 1: Initial Server Access

```bash
# SSH into your Contabo server (use credentials from email)
ssh root@YOUR_SERVER_IP

# Update system
apt update && apt upgrade -y

# Set timezone
timedatectl set-timezone Africa/Cairo  # Or your timezone
```

### Step 2: Install Required Software

```bash
# Install PHP 8.2 and extensions
apt install -y software-properties-common
add-apt-repository ppa:ondrej/php -y
apt update
apt install -y php8.2 php8.2-fpm php8.2-mysql php8.2-mbstring php8.2-xml \
    php8.2-bcmath php8.2-curl php8.2-gd php8.2-zip php8.2-redis \
    php8.2-intl php8.2-soap php8.2-imagick

# Install Nginx
apt install -y nginx

# Install MySQL
apt install -y mysql-server

# Install Redis
apt install -y redis-server

# Install Composer
curl -sS https://getcomposer.org/installer | php -- --install-dir=/usr/local/bin --filename=composer

# Install Node.js (for asset compilation)
curl -fsSL https://deb.nodesource.com/setup_20.x | bash -
apt install -y nodejs

# Install Supervisor (for queue workers)
apt install -y supervisor

# Install Certbot (for SSL)
apt install -y certbot python3-certbot-nginx
```

### Step 3: Configure MySQL

```bash
# Secure MySQL installation
mysql_secure_installation
# Answer: Y to all questions
# Set a strong root password

# Create database and user
mysql -u root -p

# In MySQL prompt:

CREATE USER 'medcapsule'@'localhost' IDENTIFIED BY 'STRONG_PASSWORD_HERE';
GRANT ALL PRIVILEGES ON medcapsule.* TO 'medcapsule'@'localhost';
FLUSH PRIVILEGES;

```

### Step 9: Configure Nginx

```bash
# Create Nginx configuration
nano /etc/nginx/sites-available/medcapsule
```

**Nginx Configuration:**

```nginx
server {
    listen 80;
    listen [::]:80;
    server_name medcapsule.xyz www.medcapsule.xyz;
    
    # Redirect to HTTPS (after SSL is configured)
    return 301 https://$server_name$request_uri;
}

server {
    listen 443 ssl http2;
    listen [::]:443 ssl http2;
    server_name medcapsule.xyz www.medcapsule.xyz;

    root /var/www/medcapsule/public;
    index index.php index.html;

    # SSL certificates (will be added by Certbot)
    ssl_certificate /etc/letsencrypt/live/medcapsule.xyz/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/medcapsule.xyz/privkey.pem;
    
    # SSL configuration
    ssl_protocols TLSv1.2 TLSv1.3;
    ssl_ciphers HIGH:!aNULL:!MD5;
    ssl_prefer_server_ciphers on;

    # Security headers
    add_header X-Frame-Options "SAMEORIGIN" always;
    add_header X-Content-Type-Options "nosniff" always;
    add_header X-XSS-Protection "1; mode=block" always;
    add_header Referrer-Policy "no-referrer-when-downgrade" always;
    add_header Content-Security-Policy "default-src 'self' http: https: data: blob: 'unsafe-inline'" always;

    # Gzip compression
    gzip on;
    gzip_vary on;
    gzip_min_length 1024;
    gzip_types text/plain text/css text/xml text/javascript application/x-javascript application/xml+rss application/json;

    # Client body size (for file uploads)
    client_max_body_size 50M;

    # Logging
    access_log /var/log/nginx/medcapsule-access.log;
    error_log /var/log/nginx/medcapsule-error.log;

    location / {
        try_files $uri $uri/ /index.php?$query_string;
    }

    location ~ \.php$ {
        fastcgi_pass unix:/var/run/php/php8.2-fpm.sock;
        fastcgi_param SCRIPT_FILENAME $realpath_root$fastcgi_script_name;
        include fastcgi_params;
        fastcgi_hide_header X-Powered-By;
        
        # Increase timeouts for long-running requests
        fastcgi_read_timeout 300;
        fastcgi_send_timeout 300;
    }

    location ~ /\.(?!well-known).* {
        deny all;
    }

    # Cache static assets
    location ~* \.(jpg|jpeg|png|gif|ico|css|js|svg|woff|woff2|ttf|eot)$ {
        expires 1y;
        add_header Cache-Control "public, immutable";
    }
}
```

```bash
# Enable site
ln -s /etc/nginx/sites-available/medcapsule /etc/nginx/sites-enabled/

# Remove default site

# Test configuration

# Restart Nginx
systemctl restart nginx
systemctl enable nginx
```

### Step 11: Setup SSL Certificate

```bash
# First, make sure your domain DNS is pointing to your server IP
# A record: medcapsule.xyz → YOUR_SERVER_IP
# A record: www.medcapsule.xyz → YOUR_SERVER_IP

# Temporarily disable SSL redirect in Nginx (comment out the redirect)
nano /etc/nginx/sites-available/medcapsule
# Comment out the return 301 line in the port 80 block

# Reload Nginx
systemctl reload nginx

# Get SSL certificate
certbot --nginx -d medcapsule.xyz -d www.medcapsule.xyz

# Follow prompts:
# - Enter email
# - Agree to terms
# - Choose to redirect HTTP to HTTPS (option 2)

# Test auto-renewal
certbot renew --dry-run

# Enable SSL redirect in Nginx (uncomment the line)
nano /etc/nginx/sites-available/medcapsule
systemctl reload nginx
```

### Step 12: Setup Queue Worker (Supervisor)

```bash
# Create supervisor configuration
nano /etc/supervisor/conf.d/medcapsule-worker.conf
```

**Supervisor Configuration:**

```ini
[program:medcapsule-worker]
process_name=%(program_name)s_%(process_num)02d
command=php /var/www/medcapsule/artisan queue:work redis --sleep=3 --tries=3 --max-time=3600
autostart=true
autorestart=true
stopasgroup=true
killasgroup=true
user=www-data
numprocs=2
redirect_stderr=true
stdout_logfile=/var/www/medcapsule/storage/logs/worker.log
stopwaitsecs=3600
```

```bash
# Update supervisor
supervisorctl reread
supervisorctl update
supervisorctl start medcapsule-worker:*

# Check status
supervisorctl status
```

## Part 4: Security Hardening

### Setup Firewall

```bash
# Install UFW
apt install -y ufw

# Allow SSH, HTTP, HTTPS
ufw allow 22/tcp
ufw allow 80/tcp
ufw allow 443/tcp

# Enable firewall
ufw enable
ufw status
```

### Disable Root SSH Login

```bash
# Create a new user
adduser medcapsule
usermod -aG sudo medcapsule

# Test SSH with new user (in a NEW terminal, don't close root session yet!)
ssh medcapsule@YOUR_SERVER_IP

# If successful, disable root login
nano /etc/ssh/sshd_config

# Change:
PermitRootLogin no
PasswordAuthentication no  # If using SSH keys

# Restart SSH
systemctl restart sshd
```

### Setup Fail2Ban

```bash
# Install Fail2Ban
apt install -y fail2ban

# Create local configuration
cp /etc/fail2ban/jail.conf /etc/fail2ban/jail.local

# Edit configuration
nano /etc/fail2ban/jail.local

# Enable SSH protection
[sshd]
enabled = true
port = 22
maxretry = 3
bantime = 3600

# Restart Fail2Ban
systemctl restart fail2ban
systemctl enable fail2ban
```

## Part 5: Monitoring & Maintenance

### Setup Automated Backups

```bash
# Create backup script
nano /root/backup-medcapsule.sh
```

**Backup Script:**

```bash
#!/bin/bash

# Configuration
BACKUP_DIR="/root/backups"
DATE=$(date +%Y%m%d_%H%M%S)
DB_NAME="medcapsule"
DB_USER="medcapsule"
DB_PASS="YOUR_DB_PASSWORD"
APP_DIR="/var/www/medcapsule"

# Create backup directory
mkdir -p $BACKUP_DIR

# Backup database
mysqldump -u $DB_USER -p$DB_PASS $DB_NAME | gzip > $BACKUP_DIR/db_$DATE.sql.gz

# Backup application files (excluding vendor and node_modules)
tar -czf $BACKUP_DIR/app_$DATE.tar.gz \
    --exclude='vendor' \
    --exclude='node_modules' \
    --exclude='storage/logs' \
    $APP_DIR

# Keep only last 7 days of backups
find $BACKUP_DIR -type f -mtime +7 -delete

echo "Backup completed: $DATE"
```

```bash
# Make executable
chmod +x /root/backup-medcapsule.sh

# Add to crontab (daily at 2 AM)
crontab -e

# Add:
0 2 * * * /root/backup-medcapsule.sh >> /var/log/backup.log 2>&1
```
